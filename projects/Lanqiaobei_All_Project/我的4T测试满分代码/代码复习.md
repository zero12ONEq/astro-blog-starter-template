# 十一届省赛第一套

```c
#include <STC15F2K60S2.H>
#include "stdio.h"
#include "led.h"
#include "init.h"
#include "key.h"
#include "seg.h"
#include "iic.h"
//按键
idata unsigned char key_val,key_old,key_down,key_up;
idata unsigned char key_slow_down;

//数码管
idata unsigned char seg_pos;
pdata unsigned char seg_buf[]={10,10,10,10,10,10,10,10};
idata unsigned char seg_slow_down;
idata unsigned char seg_disp_mode=0;//0-数据界面 1-参数设置界面 2-计数界面
//led
pdata unsigned char ucled[]={0,0,0,0,0,0,0,0};
idata unsigned int time_5s;//led 5s计时器
idata bit L1_flag;//当前电压小于电压参数标志位 1-小于
//电压值
idata unsigned char ad_slow_down;
idata unsigned int V_100x;//电压值100倍
idata unsigned int V_old_100x;//上一次的电压值100倍
idata unsigned char V_para_10x;//电压参数值10倍 0-50
//计数
idata unsigned int count=0;
idata unsigned int error_key_count;//无效按键操作次数
void key_proc()
{
	if(key_slow_down<10) return;
	key_slow_down=0;
	key_val=key_read();
	key_down=key_val&(key_val^key_old);
	key_up=~key_val&(key_val^key_old);
	key_old=key_val;
	if(key_down==12)
	{
		switch(seg_disp_mode)
		{
			case 0:
				seg_disp_mode=1;
				break;
			case 1:
				eeprom_write(&V_para_10x,0,1);
				seg_disp_mode=2;
				break;
			case 2:
				seg_disp_mode=0;
				break;
		}
	}
	
	if(key_down==13)
	{
		if(seg_disp_mode==2)
		{
			error_key_count=0;
			count=0;
		}
		else
			error_key_count++;
	}
	
		if(key_down==16)
		{
			if(seg_disp_mode==1)
			{
				error_key_count=0;
				if(V_para_10x>=50) //如果已经达到5v，那么再按下按键，回到0v
					V_para_10x=0;
				else //否则+0.5v
					V_para_10x+=5;
			}
			else
				error_key_count++;
		}
		
		if(key_down==17)
		{
			if(seg_disp_mode==1)
			{
				error_key_count=0;
				if(V_para_10x==0) //如果按下按键前已经达到0v，那么按下按键，回到5v
					V_para_10x=50;
				else
					V_para_10x-=5;
			}
			else
				error_key_count++;
		}
}

void seg_proc()
{
	if(seg_slow_down<80) return;
	seg_slow_down=0;

	switch(seg_disp_mode)
	{
		case 0: //数据界面
			seg_buf[0]=11;
			seg_buf[1]=seg_buf[2]=seg_buf[3]=seg_buf[4]=10;
			seg_buf[5]=V_100x/100%10+',';
			seg_buf[6]=V_100x/10%10;
			seg_buf[7]=V_100x %10;
			break;
		case 1: //参数设置界面
			seg_buf[0]=12;
			seg_buf[1]=seg_buf[2]=seg_buf[3]=seg_buf[4]=10;
			seg_buf[5]=V_para_10x/10%10+',';
			seg_buf[6]=V_para_10x%10;
			seg_buf[7]=0;			
			break;
		case 2: //计数界面
			seg_buf[0]=13;
			seg_buf[1]=(count>=1000000)?count/1000000%10:10;
			seg_buf[2]=(count>=100000)?count/100000%10:10;
			seg_buf[3]=(count>=10000)?count/10000%10:10;
			seg_buf[4]=(count>=1000)?count/1000%10:10;
			seg_buf[5]=(count>=100)?count/100%10:10;;
			seg_buf[6]=(count>=10)?count/10%10:10;
			seg_buf[7]=count %10;				
			break;
	}
}

void led_proc()
{
	if(V_100x<V_para_10x*10&&time_5s>5000)
	{
		L1_flag=1;
	}
	else
	{
		L1_flag=0;
	}
	ucled[0]=L1_flag;
	ucled[1]=(count%2!=0);
	ucled[2]=(error_key_count>=3);
}

void get_ad()
{
	if(ad_slow_down<100) return;
	ad_slow_down=0;
	V_old_100x=V_100x;
	ad_read(0x03);
	V_100x=(ad_read(0x03)/51.0)*100;
	if(V_old_100x>V_para_10x*10 && V_100x<V_para_10x*10)
	{
		count++;
	}	

}
void Timer1_Init(void)		//1毫秒@12.000MHz
{
	AUXR &= 0xBF;			//定时器时钟12T模式
	TMOD &= 0x0F;			//设置定时器模式
	TL1 = 0x18;				//设置定时初始值
	TH1 = 0xFC;				//设置定时初始值
	TF1 = 0;				//清除TF1标志
	TR1 = 1;				//定时器1开始计时
	ET1 = 1;				//使能定时器1中断
	EA  = 1;        //使能全局中断
}

void Timer1_Isr(void) interrupt 3
{
	key_slow_down++;
	seg_slow_down++;
	ad_slow_down++;
	if(seg_buf[seg_pos]>20)
		seg_disp(seg_pos,seg_buf[seg_pos]-',',1);
	else
		seg_disp(seg_pos,seg_buf[seg_pos],0);
	seg_pos=(++seg_pos)%8;
	if(V_100x<V_para_10x*10)
	{
		if(++time_5s==5000) time_5s=5001;
	}
	else
	{
		time_5s=0;
	}
		
	led_disp(ucled);
}
void main()
{
	system_init();
	Timer1_Init();
	eeprom_read(&V_para_10x,0,1);
	while(1)
	{
		key_proc();
		seg_proc();
		led_proc();
		get_ad();
	}
}
```

# 十二届省赛第一套

```C
#include <STC15F2K60S2.H>
#include "led.h"
#include "init.h"
#include "key.h"
#include "seg.h"
#include "onewire.h"
#include "iic.h"

//按键
idata unsigned char key_val,key_old,key_down,key_up;
idata unsigned char key_slow_down;

//数码管
idata unsigned char seg_pos;
pdata unsigned char seg_buf[]={10,10,10,10,10,10,10,10};
idata unsigned char seg_slow_down;
idata unsigned char seg_disp_mode=0;//0-温度显示界面 1-参数设置界面 2-DAC输出界面
//led
pdata unsigned char ucled[]={0,0,0,0,0,0,0,0};

//温度
idata unsigned int T_val_100x;//温度数据100倍
idata unsigned char T_para=25;//温度参数
idata unsigned char T_para_set;//温度参数设置变量
idata unsigned int temp_slow_down;
idata bit got_T_flag;
//DAC
idata unsigned int V_val_100x;//DAC输出电压值的100倍
idata bit model_flag=0;//模式标志位 0-模式1 1-模式2
void key_proc()
{
	if(key_slow_down<10) return;
	key_slow_down=0;
	key_val=key_read();
	key_down=key_val&(key_val^key_old);
	key_up=~key_val&(key_val^key_old);
	key_old=key_val;
	if(key_down==4)
	{
		switch(seg_disp_mode)
		{
			case 0:
				T_para_set=T_para;//进入设置界面前把参数值赋值给参数设置值
				seg_disp_mode=1;
				break;
			case 1:
				T_para=T_para_set;//离开参数设置界面前，把设置值赋值给参数值
				seg_disp_mode=2;
				break;
			case 2:
				seg_disp_mode=0;
				break;
		}
	}
	if(seg_disp_mode==1)
	{
		if(key_down==8) 
		{
			if(T_para_set>0)
				T_para_set--;
		}
		if(key_down==9) 
		{
			if(T_para_set<99)
				T_para_set++;
		}
	}
	if(key_down==5)
		model_flag=!model_flag;
}


void seg_proc()
{
	if(seg_slow_down<80) return;
	seg_slow_down=0;

	switch(seg_disp_mode)
	{
		case 0:
			seg_buf[0]=11;
			seg_buf[1]=seg_buf[2]=seg_buf[3]=10;
			seg_buf[4]=T_val_100x/1000%10;
			seg_buf[5]=T_val_100x/100%10+',';
			seg_buf[6]=T_val_100x/10%10;
			seg_buf[7]=T_val_100x%10;
			break;
		case 1:
			seg_buf[0]=12;
			seg_buf[1]=seg_buf[2]=seg_buf[3]=seg_buf[4]=seg_buf[5]=10;
			seg_buf[6]=T_para_set/10%10;
			seg_buf[7]=T_para_set%10;
			break;
		case 2:
			seg_buf[0]=13;
			seg_buf[1]=seg_buf[2]=seg_buf[3]=seg_buf[4]=10;
			seg_buf[5]=V_val_100x/100%10+',';
			seg_buf[6]=V_val_100x/10%10;
			seg_buf[7]=V_val_100x%10;			
			break;
	}
}


void led_proc()
{
	ucled[0]=(!model_flag);
	ucled[1]=(seg_disp_mode==0);
	ucled[2]=(seg_disp_mode==1);
	ucled[3]=(seg_disp_mode==2);
}


void get_temperature()
{
	float temp;
	if(temp_slow_down<300) return;
	temp_slow_down=0;
	temp=rd_temperature();
	T_val_100x=(unsigned int)(temp*100);
	got_T_flag=1;
}

void da_proc()
{
	if(got_T_flag)
	{
		got_T_flag=0;
		if(model_flag==0)
		{
			if(T_val_100x<T_para*100)
				V_val_100x=0;
			else
				V_val_100x=500;
		}
		else
		{
			if(T_val_100x>=4000)
				V_val_100x=400;
			else if(T_val_100x>2000)
				V_val_100x=(unsigned int)((((3.0/20.0)*(T_val_100x/100.0-20))+1.0)*100);
			else
				V_val_100x=100;
		}
		da_write((float)V_val_100x/100.0*51.0);
	}
}
void Timer1_Init(void)		//1毫秒@12.000MHz
{
	AUXR &= 0xBF;			//定时器时钟12T模式
	TMOD &= 0x0F;			//设置定时器模式
	TL1 = 0x18;				//设置定时初始值
	TH1 = 0xFC;				//设置定时初始值
	TF1 = 0;				//清除TF1标志
	TR1 = 1;				//定时器1开始计时
	ET1 = 1;				//使能定时器1中断
	EA  = 1;        //使能全局中断
}

void Timer1_Isr(void) interrupt 3
{
	key_slow_down++;
	seg_slow_down++;
	temp_slow_down++;
	if(seg_buf[seg_pos]>20)
		seg_disp(seg_pos,seg_buf[seg_pos]-',',1);
	else
		seg_disp(seg_pos,seg_buf[seg_pos],0);
	seg_pos=(++seg_pos)%8;
	
	led_disp(ucled);
}


void main()
{
	system_init();
	Timer1_Init();
	while(1)
	{
		key_proc();
		seg_proc();
		led_proc();
		get_temperature();
		da_proc();
	}
}
```

# 十三届国赛

```c
#include <STC15F2K60S2.H>
#include "led.h"
#include "init.h"
#include "key.h"
#include "seg.h"
#include "iic.h"
#include "ultrasound.h"
//频率
idata unsigned int freq;
idata unsigned int freq_time_1s;
idata unsigned char freq_para_10x=90;//(10-120)1.0-12.0 单位kHz
//按键
idata unsigned char key_val,key_old,key_down,key_up;
idata unsigned char key_slow_down;
idata bit key_long_flag;
idata unsigned int key_long_tick;
//数码管
idata unsigned char seg_pos;
pdata unsigned char seg_buf[]={10,10,10,10,10,10,10,10};
idata unsigned char seg_slow_down;
idata unsigned char seg_disp_mode=0;//0-频率界面 1-湿度界面 2-测距界面 3-参数界面
idata bit freq_danwei=0;//0-Hz 1-kHz
idata bit dist_danwei=0;//0-cm 1-m
idata unsigned char para_disp_mode;//0-频率参数 1-湿度参数 2-距离参数
//led
pdata unsigned char ucled[]={0,0,0,0,0,0,0,0};
idata bit L1_flash_flag;
idata bit L2_flash_flag;
idata bit L3_flash_flag;
idata unsigned char led1_flash_tick;
idata unsigned char led2_flash_tick;
idata unsigned char led3_flash_tick;
//继电器
idata unsigned char relay_count;
idata bit relay_old = 0;
idata bit relay_count_dirty = 0;
//电机
idata unsigned char motor_period;//0-9
idata unsigned char motor_compare;
//ad
idata unsigned char ad_slow_down;
//超声波
idata unsigned char dist_slow_down;
idata unsigned char dist_val;
idata unsigned char dist_para_10x=6;//单位m (0.1-1.2)1-12
//湿度
idata unsigned char W_val;
idata unsigned char W_para=40;//10-60 单位：百分之
//eeprom
idata unsigned char relay_count_pre = 0;
idata unsigned char eeprom_lock = 13;
idata unsigned char eeprom_temp = 0;
void key_proc()
{
	if(key_slow_down<10) return;
	key_slow_down=0;
	key_val=key_read();
	key_down=key_val&(key_val^key_old);
	key_up=~key_val&(key_val^key_old);
	key_old=key_val;
	if(key_down==4)
	{
		seg_disp_mode=(++seg_disp_mode)%4;
		para_disp_mode=0;
		led_off();
	}
	if(seg_disp_mode==3)
	{
		if(key_down==5)
		{
			para_disp_mode=(++para_disp_mode)%3;
		}
		if(key_down==6)
		{
			if(para_disp_mode==0)
			{
				if(freq_para_10x>=120) 
					freq_para_10x=10;
				else
					freq_para_10x+=5;
			}
			else if(para_disp_mode==1)
			{
				if(W_para>=60) 
					W_para=10;
				else	
					W_para+=10;
			}
			else
			{
				if(dist_para_10x>=12) 
					dist_para_10x=1;
				else
					dist_para_10x++;
			}
		}
		
		if(key_down==7)
		{
			if(para_disp_mode==0)
			{
				if(freq_para_10x<=10) 
					freq_para_10x=120;
				else
					freq_para_10x-=5;
			}
			else if(para_disp_mode==1)
			{
				if(W_para<=10) 
					W_para=60;
				else
					W_para-=10;
			}
			else
			{
				if(dist_para_10x<=1) 
					dist_para_10x=12;
				else
					dist_para_10x--;
			}
		}		
	}
	if(seg_disp_mode==2)
	{
		if(key_down==6)
			dist_danwei=!dist_danwei;
	}
	if(seg_disp_mode==0)
	{
		if(key_down==7)
			freq_danwei=!freq_danwei;
	}
	if(seg_disp_mode==1)
	{
		if(key_down==7)
		{
			key_long_flag=1;
		}
		if(key_up==7)
		{
			if(key_long_tick>=1000)
			{
				relay_count=0;
				relay_count_dirty = 1;
			}
			key_long_flag=0;
			key_long_tick=0;
		}
	}
}


void seg_proc()
{
	if(seg_slow_down<80) return;
	seg_slow_down=0;
	switch(seg_disp_mode)
	{
		case 0:
			seg_buf[0]=11;
			seg_buf[1]=10;
			if(freq_danwei==0)
			{
				seg_buf[2]=(freq>=100000)?freq/100000%10:10;
				seg_buf[3]=(freq>=10000)?freq/10000%10:10;
				seg_buf[4]=(freq>=1000)?freq/1000%10:10;
				seg_buf[5]=(freq>=100)?freq/100%10:10;
				seg_buf[6]=(freq>=10)?freq/10%10:10;
				seg_buf[7]=freq%10;
			}
			else
			{
				seg_buf[2]=10;
				seg_buf[3]=10;
				seg_buf[4]=(freq>=100000)?freq/100000%10:10;
				seg_buf[5]=(freq>=10000)?freq/10000%10:10;
				seg_buf[6]=freq/1000%10+',';
				seg_buf[7]=freq/100%10;				
			}
			break;
		case 1:
			seg_buf[0]=12;
			seg_buf[1]=seg_buf[2]=seg_buf[3]=seg_buf[4]=seg_buf[5]=10;
			seg_buf[6]=W_val/10%10;
			seg_buf[7]=W_val%10;
			break;
		case 2:
			seg_buf[0]=13;
			seg_buf[1]=seg_buf[2]=seg_buf[3]=seg_buf[4]=10;
			if(dist_danwei==0)
			{
				seg_buf[5]=(dist_val>=100)?dist_val/100%10:10;
				seg_buf[6]=(dist_val>=10)?dist_val/10%10:10;
				seg_buf[7]=dist_val%10;
			}
			else
			{
				seg_buf[5]=dist_val/100%10+',';
				seg_buf[6]=dist_val/10%10;
				seg_buf[7]=dist_val%10;				
			}
			break;
		case 3:
			seg_buf[0]=14;
			if(para_disp_mode==0)
			{
				seg_buf[1]=1;
				seg_buf[2]=seg_buf[3]=seg_buf[4]=10;
				seg_buf[5]=(freq_para_10x>=100)?freq_para_10x/100%10:10;
				seg_buf[6]=freq_para_10x/10%10+',';
				seg_buf[7]=freq_para_10x%10;
			}
			else if(para_disp_mode==1)
			{
				seg_buf[1]=2;
				seg_buf[2]=seg_buf[3]=seg_buf[4]=seg_buf[5]=10;
				seg_buf[6]=W_para/10%10;
				seg_buf[7]=W_para%10;				
			}
			else
			{
				seg_buf[1]=3;
				seg_buf[2]=seg_buf[3]=seg_buf[4]=seg_buf[5]=10;
				seg_buf[6]=dist_para_10x/10%10+',';
				seg_buf[7]=dist_para_10x%10;			
			}
			break;		
	
	}
}

void led_proc()
{
	switch(seg_disp_mode) //0-频率界面 1-湿度界面 2-测距界面 3-参数界面
	{
		case 0:
			ucled[0]=1;
			ucled[1]=0;
			ucled[2]=0;
			break;
		case 1:
			ucled[0]=0;
			ucled[1]=1;
			ucled[2]=0;
			break;
		case 2:
			ucled[0]=0;
			ucled[1]=0;			
			ucled[2]=1;
			break;
		case 3: //0-频率参数 1-湿度参数 2-距离参数
			if(para_disp_mode==0)
			{
				ucled[0]=L1_flash_flag;
				ucled[1]=0;
				ucled[2]=0;
			}
			else if(para_disp_mode==1)
			{
				ucled[1]=L2_flash_flag;
				ucled[0]=ucled[2]=0;
			}
			else
			{
				ucled[0]=ucled[1]=0;
				ucled[2]=L3_flash_flag;
			}
			break;
	}
	ucled[3]=(freq>freq_para_10x*100);
	ucled[4]=(W_val>W_para);
	ucled[5]=(dist_val>dist_para_10x*10);
}



void relay_proc()
{
	bit relay_new = (dist_val > dist_para_10x * 10);

	if(relay_new != relay_old)
	{
			relay(relay_new);
			relay_old = relay_new;
			relay_count++;
			relay_count_dirty = 1;
	}
}
		
void eeprom_proc()
{
	if(!relay_count_dirty) return;
	eeprom_write(&relay_count, 0, 1);
	relay_count_dirty = 0;
}

void motor_proc()
{
	if(freq>freq_para_10x*100)
		motor_compare=8;//占空比为8
	else
		motor_compare=2;//占空比为2
}
void ad_proc()
{
	float temp1;
	float temp2;
	if(ad_slow_down<100) return;
	ad_slow_down=0;
	ad_read(0x43);
	temp1=ad_read(0x43)/51.0;
	if(temp1>5)
		W_val=100;
	else
		W_val=(unsigned char)(20*temp1);

	if(W_val>80)
		da_write(5*51);
	else if(W_val>=W_para)
	{
		temp2=(4.0/(float)(80-W_para))*(W_val-W_para)+1.0;
		da_write(temp2*51);
	}
	else
	{
		da_write(1*51);
	}
}


void dist_proc()
{
	if(dist_slow_down<100) return;
	dist_slow_down=0;
	dist_val=ut_wave_data();
}

void Timer0_Init(void)		//1毫秒@12.000MHz
{
	AUXR &= 0x7F;			//定时器时钟12T模式
	TMOD &= 0xF0;			//设置定时器模式
	TMOD |= 0x05;			//设置定时器模式 为P34引脚读取外部脉冲模式
```

```c
/* ========== 刷新速度值（用当前R/B/freq计算Speed_10x）========== */
void Refresh_Speed()
{
    float temp_speed_10x;
    temp_speed_10x = ((float)(PI * (Para_R_10x / 10.0) * freq) / 100.0f + Para_B) * 10;
    Speed_10x = (temp_speed_10x < 0) ? 0 : (unsigned int)temp_speed_10x;
}

/* ========== 重置运动状态（锚定起点+重置时间+重算速度分量）========== */
void Reset_Movement()
{
    float dx_f, dy_f, dist;

    Start_Location[0] = Current_Location[0];
    Start_Location[1] = Current_Location[1];
    Movement_Time = 0;
    cnt_movement_time = 0;

    dx_f = (float)Destination[0] - (float)Current_Location[0];
    dy_f = (float)Destination[1] - (float)Current_Location[1];
    dist = sqrt(dx_f * dx_f + dy_f * dy_f);
    if (dist > 0)
    {
        speed_x = ((float)Speed_10x / 10.0f) * dx_f / dist;
        speed_y = ((float)Speed_10x / 10.0f) * dy_f / dist;
    }
    else
    {
        speed_x = 0;
        speed_y = 0;
    }
}

/* ========== 控制函数（线性插值，200ms调度）========== */
void Ctrl_Proc()
{
    if(ctrl_slow_down<200) return;
    ctrl_slow_down=0;
    if (State_Disp_Mode == 1)
    {
        float new_x, new_y;
        float t_sec = Movement_Time * 0.1f;

        // 障碍物保护
        if (Barrier_Distance <= 30)
        {
            State_Disp_Mode = 2;
            return;
        }

        // 线性插值计算当前位置
        new_x = (float)Start_Location[0] + speed_x * t_sec;
        new_y = (float)Start_Location[1] + speed_y * t_sec;

        // 到达检测
        if (speed_x >= 0)
        {
            if (new_x >= (float)Destination[0]) new_x = (float)Destination[0];
        }
        else
        {
            if (new_x <= (float)Destination[0]) new_x = (float)Destination[0];
        }
        if (speed_y >= 0)
        {
            if (new_y >= (float)Destination[1]) new_y = (float)Destination[1];
        }
        else
        {
            if (new_y <= (float)Destination[1]) new_y = (float)Destination[1];
        }

        Current_Location[0] = (unsigned int)(new_x + 0.5f);
        Current_Location[1] = (unsigned int)(new_y + 0.5f);

        // 两轴都到达目的地
        if (Current_Location[0] == Destination[0] && Current_Location[1] == Destination[1])
        {
            State_Disp_Mode = 0;
            Reach_Flag = 1;
            Time_3s = 0;
        }
    }
}
```

```c
		Distance_Val=Ut_Wave_Data();

		if(!State_Change_Flag)
		{
			if(Distance_Sub<5) temp_motion_state=0;
```

```c
#include "led.h"

idata unsigned char key_val, key_old, key_down, key_up;
idata unsigned char key_slow_down;

/* ============ 数码管 ============ */
idata unsigned char seg_pos;
pdata unsigned char seg_buf[] = {10,10,10,10,10,10,10,10};
idata unsigned char seg_slow_down;

/* ============ LED ============ */
pdata unsigned char ucled[] = {0,0,0,0,0,0,0,0};
idata unsigned char led_flash_tick;
bit led_flash_flag = 0;

/* ============ 界面模式 ============ */

idata unsigned char range_show_mode = 0;  // 0-初始/清除 1-单次 2-累加
idata unsigned char para_disp_mode = 0;   // 0-时间参数 1-次数参数

/* ============ 温度与声速 ============ */
idata unsigned char T_val;
idata unsigned int T_slow_down;
idata float speed = 340;

/* ============ 测距 ============ */
idata unsigned int dist_val;       // 单次测量结果
idata unsigned int dist_sum;       // 累加测量累计结果
bit signal_flag = 0;               // 单次测量待执行
bit summation_flag = 0;            // 累加测量执行中

/* ============ 参数 ============ */
pdata unsigned char time_apart[4] = {2, 4, 6, 4};
idata unsigned char time_apart_index = 0;
pdata unsigned char dist_count[4] = {3, 5, 7, 5};
idata unsigned char dist_count_index = 0;
idata unsigned char summation_index = 0;  // 累加已完成次数

/* ============ 累加测量定时 ============ */
idata unsigned int dist_test_timer;

/* ================================================
 *  测距执行（单次/累加共用）
 *  被 key_proc 和 dist_proc 调用
 * ================================================ */
void do_distance()
{
	unsigned int temp;
	if(signal_flag)
	{
		temp = ut_wave_data(speed);
		dist_val = temp;
		signal_flag = 0;  // 单次测量完成
	}
	else if(summation_flag)
	{
		temp = ut_wave_data(speed);
		dist_sum += temp;
		summation_index++;
		if(summation_index >= dist_count[dist_count_index])
		{
			summation_flag = 0;   // 累加测量全部完成
			summation_index = 0;
		}
	}
}

/* ================================================
 *  按键处理
 * ================================================ */
void key_proc()
{
	if(key_slow_down < 10) return;
	key_slow_down = 0;

	key_val = key_read();
	key_down = key_val & (key_val ^ key_old);
	key_up = ~key_val & (key_val ^ key_old);
	key_old = key_val;

	/* --- 累加测量执行中: S8/S12 无效 --- */
	if(summation_flag == 0)
	{
		// S8: 界面切换
		if(key_down == 8)
		{
			if(seg_disp_mode == 0)
			{
				seg_disp_mode = 1;
				para_disp_mode = 0;
			}
			else
			{
				if(para_disp_mode == 0)
					para_disp_mode = 1;
				else
					seg_disp_mode = 0;
			}
		}
		// S12: 清除（仅测距界面）
		if(seg_disp_mode == 0 && key_down == 12)
		{
			dist_val = 0;
			dist_sum = 0;
			range_show_mode = 0;  // 回到初始显示
		}
	}

	/* --- 测距界面下: S4/S5 --- */
	if(seg_disp_mode == 0)
	{
		// S4: 单次测量（可立即中断累加）
		if(key_down == 4)
		{
			signal_flag = 1;
			summation_flag = 0;
			summation_index = 0;
			dist_val = 0;
			dist_sum = 0;
			range_show_mode = 1;
			do_distance();
		}
		// S5: 累加测量
		if(key_down == 5)
		{
			summation_flag = 1;
			signal_flag = 0;
			summation_index = 0;
			dist_val = 0;
			dist_sum = 0;
			dist_test_timer = 0;
			range_show_mode = 2;
			do_distance();  // 立即执行首次测距
		}
	}

	/* --- 参数设置界面: S9 --- */
	if(seg_disp_mode == 1 && key_down == 9)
	{
		if(para_disp_mode == 0)
			time_apart_index = (++time_apart_index) % 4;
		else
			dist_count_index = (++dist_count_index) % 4;
	}
}

/* ================================================
 *  数码管显示
 * ================================================ */
void seg_proc()
{
	if(seg_slow_down < 80) return;
	seg_slow_down = 0;

	switch(seg_disp_mode)
	{
		case 0: // 测距界面
			seg_buf[1] = 12; // C
			seg_buf[2] = T_val / 10 % 10;
			seg_buf[3] = T_val % 10;
			switch(range_show_mode)
			{
				case 0: // 初始/清除 -> 熄灭
					seg_buf[0] = 12; // C
					seg_buf[4] = 10;
					seg_buf[5] = 10;
					seg_buf[6] = 10;
					seg_buf[7] = 10;
					break;
				case 1: // 单次 -> A
					seg_buf[0] = 11;
					seg_buf[4] = (dist_val >= 1000) ? dist_val/1000%10 : 10;
					seg_buf[5] = (dist_val >= 100)  ? dist_val/100%10  : 10;
					seg_buf[6] = (dist_val >= 10)   ? dist_val/10%10   : 10;
					seg_buf[7] = dist_val % 10;
					break;
				case 2: // 累加 -> E
					seg_buf[0] = 13;
					seg_buf[4] = (dist_sum >= 1000) ? dist_sum/1000%10 : 10;
					seg_buf[5] = (dist_sum >= 100)  ? dist_sum/100%10  : 10;
					seg_buf[6] = (dist_sum >= 10)   ? dist_sum/10%10   : 10;
					seg_buf[7] = dist_sum % 10;
					break;
			}
			break;

		case 1: // 参数设置界面
			seg_buf[0] = 14; // P
			seg_buf[2] = 10;
			seg_buf[3] = 10;
			seg_buf[4] = 10;
			seg_buf[5] = 10;
			seg_buf[6] = 0;  // 高位补0
			if(para_disp_mode == 0)
			{
				seg_buf[1] = 1;
				seg_buf[7] = time_apart[time_apart_index];
			}
			else
			{
				seg_buf[1] = 2;
				seg_buf[7] = dist_count[dist_count_index];
			}
			break;
	}
}

void led_proc()

	{

		case 1: // 单次完成后判断

			break;
		case 2: // 累加完成后判断，执行中状态不变
			if(summation_flag == 0)
				ucled[1] = (dist_sum < 20 || dist_sum > 100);
			break;
	}

void da_proc()
{
	float temp = 0;
	switch(range_show_mode)
	{
		case 0: // 未完成有效测量 -> 0V
			temp = 0;
			break;
		case 1: // 单次测量完成后更新
			if(signal_flag == 0)
			{
				if(dist_val <= 20)       temp = 1;
				else if(dist_val >= 100) temp = 5;
				else                     temp = (float)dist_val / 20.0;
			}
			break;
		case 2: // 累加测量完成后更新
			if(summation_flag == 0)
			{
				if(dist_sum <= 20)       temp = 1;
				else if(dist_sum >= 100) temp = 5;
				else                     temp = (float)dist_sum / 20.0;
			}
			break;
	}
	da_write((unsigned char)(temp * 51));
}
```
